Progress store (persisted)

Install Zustand: npm i zustand

Create src/store/progress.ts:

import { create } from "zustand";

type Progress = {
  solved: Record<string, boolean>;
  digits: number[];         // e.g., [4,6,2]
  tokens: Record<string, boolean>; // e.g., { UNLOCK5: true }
  markSolved: (args: { id: string; digit?: number; token?: string }) => void;
  reset: () => void;
};

const STORAGE_KEY = "glitch_progress_v1";

function load(): Omit<Progress, "markSolved"|"reset"> {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return { solved: {}, digits: [], tokens: {} };
    const obj = JSON.parse(raw);
    return { solved: obj.solved ?? {}, digits: obj.digits ?? [], tokens: obj.tokens ?? {} };
  } catch { return { solved: {}, digits: [], tokens: {} }; }
}

export const useProgress = create<Progress>((set, get) => ({
  ...load(),
  markSolved: ({ id, digit, token }) => {
    const next = { ...get().solved, [id]: true };
    const digits = [...get().digits];
    if (typeof digit === "number" && !digits.includes(digit)) digits.push(digit);
    const tokens = { ...get().tokens };
    if (token) tokens[token] = true;
    const snapshot = { solved: next, digits, tokens };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(snapshot));
    set(snapshot);
  },
  reset: () => {
    const snapshot = { solved: {}, digits: [], tokens: {} };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(snapshot));
    set(snapshot);
  }
}));

Small helpers

Create src/lib/puzzles.ts with utilities:

import defs from "@/../content/puzzles/index.json";
export const allPuzzles = defs;
export const byId = (id: string) => allPuzzles.find(p => p.id === id);
export const unmetDeps = (id: string, solved: Record<string, boolean>) =>
  (byId(id)?.requires ?? []).filter(dep => !solved[dep]);
